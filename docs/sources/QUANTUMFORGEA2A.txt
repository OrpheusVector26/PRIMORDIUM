# 3. HANDSHAKE — WHAT A HANDSHAKE ACTUALLY IS HERE

`[fetched]` A2A has **no dedicated handshake RPC**. There is no `hello`, no `negotiate`, no version-exchange method. The handshake is a composition of three ordinary operations:

1. `GET /.well-known/agent-card.json` — unauthenticated discovery.
2. Verify the card's JWS signature over its RFC 8785 canonical form.
3. `SendMessage` with `role: ROLE_USER` carrying a typed intent envelope.

`[fetched]` A2A v1.0 moved `url`, `preferredTransport`, `additionalInterfaces` and top-level `protocolVersion` into a single ordered `supportedInterfaces` array; `security` became `securityRequirements`; `supportsAuthenticatedExtendedCard` became `capabilities.extendedAgentCard`; `capabilities.stateTransitionHistory` was removed. A card written against v0.x will look unauthenticated to a v1.0 client, because the old `security` field is simply ignored.

Task states, `[fetched]`: `SUBMITTED`, `WORKING`, `COMPLETED`, `FAILED`, `CANCELED`, `REJECTED`, `INPUT_REQUIRED`, `AUTH_REQUIRED`. Parts are exactly-one-of `text` / `raw` / `url` / `data`. Results SHOULD come back as **artifacts**, not messages.

```mermaid
sequenceDiagram
    autonumber
    participant C as Client Agent
    participant W as well-known URI
    participant S as QuantumForge /a2a/v1

    C->>W: GET /.well-known/agent-card.json
    W-->>C: 200 AgentCard (unsigned or JWS-signed)
    C->>C: Validate required fields
    C->>C: Canonicalize RFC 8785, verify >=1 JWS
    C->>C: Intersect defaultInputModes with own modes
    C->>C: Match requested tags against skills[]
    alt No mode or skill overlap
        C->>C: Abort, no task created
    else Overlap confirmed
        C->>S: SendMessage role=ROLE_USER, data part = intent envelope
        S->>T: Create task, state SUBMITTED
        S-->>C: Task SUBMITTED (or WORKING via SSE)
        alt Missing parameter
            S-->>C: state INPUT_REQUIRED + question artifact
            C->>S: SendMessage with taskId, ROLE_USER
        else Sufficient
            S->>T: state WORKING
            S-->>C: SendStreamingMessage SSE status deltas
            S->>T: state COMPLETED
            S-->>C: Artifact parts (code, verdict, limits)
        end
    end
```

        raise CardTransportError(
            f"{url} is plain http; discovery is unauthenticated in transit "
            f"(pass --allow-http to accept)"
        )

    with _client(timeout) as client:
        resp = client.get(url, headers={"Accept": "application/json"})

    final_scheme = urlsplit(str(resp.url)).scheme
    if final_scheme == "http" and not allow_http:
        raise CardTransportError(f"redirected to plain http at {resp.url}")

    if resp.status_code != 200:
        raise CardTransportError(
            f"GET {url} -> {resp.status_code} {resp.reason_phrase}; "
            f"body starts {resp.text[:120]!r}"
        )
    ctype = resp.headers.get("content-type", "")
    if "json" not in ctype:
        raise CardTransportError(
            f"GET {url} returned content-type {ctype!r}, expected JSON; "
            f"body starts {resp.text[:120]!r}"
        )
    try:
        card = resp.json()
    except json.JSONDecodeError as exc:
        raise CardTransportError(f"GET {url} returned unparseable JSON: {exc}") from exc
    if not isinstance(card, dict):
        raise CardTransportError(f"card root is {type(card).__name__}, expected object")
    return card

def detect_legacy_card(card: dict[str, Any]) -> None:
    hits = [k for k in LEGACY_MARKERS if k in card]
    if not hits:
        return
    hints = "; ".join(f"{k} -> {LEGACY_MARKERS[k]}" for k in hits)
    raise LegacyCardError(
        f"card carries v0.x top-level field(s) {hits}; v1.0 migration: {hints}. "
        f"This client speaks v1.0 only."
    )


def validate_card(card: dict[str, Any]) -> list[str]:
    """Return a list of schema defects rather than raising on the first."""
    defects: list[str] = []

    missing = [f for f in REQUIRED_CARD_FIELDS if f not in card]
    if missing:
        defects.append(f"missing required fields: {missing}")
        return defects  # downstream checks need these present

    if not isinstance(card["supportedInterfaces"], list) or not card["supportedInterfaces"]:
        defects.append("supportedInterfaces is empty or not a list")
    else:
        for idx, iface in enumerate(card["supportedInterfaces"]):
            if not isinstance(iface, dict):
                defects.append(f"supportedInterfaces[{idx}] is not an object")
                continue
            for req in ("url", "protocolBinding", "protocolVersion"):
                if req not in iface:
                    defects.append(f"supportedInterfaces[{idx}] missing {req!r}")

    if not isinstance(card["skills"], list) or not card["skills"]:
        defects.append("skills is empty or not a list")
    else:
        for idx, skill in enumerate(card["skills"]):
            if not isinstance(skill, dict):
                defects.append(f"skills[{idx}] is not an object")
                continue
            for req in ("id", "name", "description", "tags"):
                if req not in skill:
                    defects.append(f"skills[{idx}] missing {req!r}")
            if "tags" in skill and not isinstance(skill["tags"], list):
                defects.append(f"skills[{idx}].tags is not a list")

    for mode_field in ("defaultInputModes", "defaultOutputModes"):
        modes = card.get(mode_field)
        if not isinstance(modes, list) or not modes:
            defects.append(f"{mode_field} is empty or not a list")

    caps = card.get("capabilities")
    if not isinstance(caps, dict):
        defects.append("capabilities is not an object")

    return defects


def verify_card_signature(card: dict[str, Any], jwks_doc: dict[str, Any]) -> None:
    """Verify >=1 JWS over the RFC 8785 canonical form of the card minus 'signatures'."""
    sigs = card.get("signatures")
    if not sigs:
        raise CardSignatureError("card is unsigned")

    body = {k: v for k, v in card.items() if k != "signatures"}
    canonical = rfc8785.dumps(body)
    key_set = jwk.JWKSet.from_json(json.dumps(jwks_doc))

    failures: list[str] = []
    for entry in sigs:
        try:
            protected = entry["protected"]
            token = jws.JWS()
            token.deserialize(json.dumps({
                "protected": protected,
                "payload": "",
                "signature": entry["signature"],
            }))
            kid = json.loads(_b64url_decode(protected)).get("kid")
            # UNVERIFIED: jwk.JWKSet.get_key(None) behaviour when no kid is present
            candidate = key_set.get_key(kid) if kid else key_set.get_key(None)
            token.verify(candidate, detached_payload=canonical)
            return
        except Exception as exc:  # noqa: BLE001 - collect every failure for the report
            failures.append(f"{type(exc).__name__}: {exc}")

    raise CardSignatureError("no signature verified: " + "; ".join(failures))


def check_card(card: dict[str, Any], jwks_doc: dict[str, Any] | None) -> None:
    """Run schema and signature passes independently; report every failure."""
    schema_defects = validate_card(card)
    sig_error: str | None = None
    if jwks_doc is not None:
        try:
            verify_card_signature(card, jwks_doc)
        except CardSignatureError as exc:
            sig_error = str(exc)

    if schema_defects or sig_error:
        parts = []
        if schema_defects:
            parts.append("schema: " + " | ".join(schema_defects))
        if sig_error:
            parts.append("signature: " + sig_error)
        raise CardSchemaError("; ".join(parts))


# --------------------------------------------------------------------- negotiation

def select_interface(card: dict[str, Any]) -> dict[str, Any]:
    """Ordered scan. Skip entries the client cannot speak; fail only if none fit."""
    skipped: list[str] = []
    for idx, iface in enumerate(card["supportedInterfaces"]):
        if not isinstance(iface, dict):
            skipped.append(f"[{idx}] not-an-object")
            continue
        version = iface.get("protocolVersion")
        if version != PROTOCOL_VERSION:
            skipped.append(f"[{idx}] protocolVersion={version!r}")
            continue
        binding = iface.get("protocolBinding")
        if binding not in SUPPORTED_BINDINGS:
            skipped.append(f"[{idx}] protocolBinding={binding!r}")
            continue
        return iface
    raise ProtocolVersionError(
        f"no interface speaks protocolVersion {PROTOCOL_VERSION!r} over "
        f"{sorted(SUPPORTED_BINDINGS)}; skipped: {skipped or 'nothing'}"
    )


def maybe_fetch_extended_card(
    card: dict[str, Any], endpoint: str, token: str, timeout: float | None = None
) -> dict[str, Any]:
    """Fetch the authenticated extended card when advertised; else return the public one."""
    if not card.get("capabilities", {}).get("extendedAgentCard", False):
        return card

    payload = {
        "jsonrpc": "2.0",
        "id": str(uuid.uuid4()),
        # UNVERIFIED: GetExtendedAgentCard params shape; empty object assumed
        "method": "GetExtendedAgentCard",
        "params": {},
    }
    with _client(timeout) as client:
        resp = client.post(
            endpoint, json=payload,
            headers={"Authorization": f"Bearer {token}", "Accept": "application/json"},
        )
    if resp.status_code != 200:
        return card
    body = resp.json()
    if "error" in body:
        return card
    extended = body.get("result")
    if not isinstance(extended, dict):
        return card
    merged = dict(card)
    merged["skills"] = extended.get("skills", card["skills"])
    return merged


def negotiate(
    card: dict[str, Any], want_tags: set[str], streaming: bool
) -> dict[str, Any]:
    matched = [
        s["id"] for s in card["skills"]
        if want_tags & set(s.get("tags") or [])
    ]
    if not matched:
        offered = sorted({t for s in card["skills"] for t in (s.get("tags") or [])})
        raise NegotiationError(
            f"no skill matches tags {sorted(want_tags)}; offered tags: {offered}"
        )

    if streaming and not card.get("capabilities", {}).get("streaming", False):
        raise NegotiationError(
            "streaming requested but capabilities.streaming is false"
        )

    in_modes = [m for m in card["defaultInputModes"] if m in ("text/plain", "application/json")]
    if not in_modes:
        raise NegotiationError(
            f"no input mode overlap; agent accepts {card['defaultInputModes']}"
        )

    return {
        "matched_skills": matched,
        "input_modes": in_modes,
        "output_modes": list(card["defaultOutputModes"]),
        "streaming": streaming,
    }

# ------------------------------------------------------------------------ dispatch

def _rpc_payload(method: str, envelope: dict[str, Any]) -> dict[str, Any]:
    message_id = str(uuid.uuid4())
    return {
        "jsonrpc": "2.0",
        "id": message_id,
        "method": method,
        "params": {
            "message": {
                "messageId": message_id,
                "role": "ROLE_USER",
                "parts": [{"data": envelope, "mediaType": "application/json"}],
            }
        },
    }


def _unwrap(body: dict[str, Any]) -> dict[str, Any]:
    if "error" in body:
        err = body["error"]
        raise JsonRpcError(err.get("code"), err.get("message", "<no message>"))
    result = body.get("result")
    if not isinstance(result, dict):
        raise JsonRpcError(None, f"result is {type(result).__name__}, expected object")
    return result


def send_message(
    endpoint: str, envelope: dict[str, Any], timeout: float | None = None
) -> dict[str, Any]:
    with _client(timeout) as client:
        resp = client.post(
            endpoint, json=_rpc_payload("SendMessage", envelope),
            headers={"Accept": "application/json"},
        )
    if resp.status_code != 200:
        raise CardTransportError(
            f"POST {endpoint} -> {resp.status_code}; body starts {resp.text[:120]!r}"
        )
    return _unwrap(resp.json())


def stream_message(
    endpoint: str, envelope: dict[str, Any], timeout: float | None = None
) -> Iterator[dict[str, Any]]:
    """Yield each SSE event's result object until the stream closes."""
    with _client(timeout) as client:
        with client.stream(
            "POST", endpoint, json=_rpc_payload("SendStreamingMessage", envelope),
            headers={"Accept": "text/event-stream"},
        ) as resp:
            if resp.status_code != 200:
                resp.read()
                raise CardTransportError(
                    f"POST {endpoint} -> {resp.status_code}; body starts {resp.text[:120]!r}"
                )
            data: list[str] = []
            for line in resp.iter_lines():
                if line == "":
                    if data:
                        raw = "\n".join(data)
                        data.clear()
                        yield _unwrap(json.loads(raw))
                    continue
                if line.startswith(":"):
                    continue
                if line.startswith("data:"):
                    data.append(line[5:].lstrip())
            if data:
                yield _unwrap(json.loads("\n".join(data)))


def _state_of(result: dict[str, Any]) -> str:
    return (result.get("status") or {}).get("state", "TASK_STATE_UNSPECIFIED")
